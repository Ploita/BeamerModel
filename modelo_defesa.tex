\begin{frame}{Objetivos da pesquisa}
\begin{itemize}
    \item Objetivo geral: aprofundar a compreensão da dinâmica do aprendizado em RL a partir do plano de informação.
    \item Investigar o fluxo de informação na rede profunda durante o aprendizado do agente.
    \item Analisar trajetórias no plano de informação e identificar possíveis estágios do aprendizado.
    \item Propor variação do plano de informação para RL sem depender de rótulo explícito.
    \item Avaliar efeito de topologias distintas em diferentes ambientes.
\end{itemize}
\end{frame}

\section{Metodologia}
\begin{frame}{Metodologia proposta}
\begin{enumerate}
    \item Agente PPO com topologia padrão de duas camadas intermediárias (32 neurônios cada).
    \item Topologias alternativas com uma e três camadas para comparação.
    \item Estimação de informação mútua com NPEET/KNN \parencite{kraskov2004estimating}.
    \item Ambientes: Pendulum, Acrobot, MountainCar e CartPole.
    \item Cada experimento executado por $10^6$ passos.
\end{enumerate}
\end{frame}

\begin{frame}{Metodologia}{Fluxo de treinamento e estimação}
\begin{itemize}
    \item Interação com ambiente até preencher buffer de 2048 passos.
    \item Pausa da coleta e atualização das redes ator/crítico por 10 épocas.
    \item Estimação de informação mútua a cada época de atualização.
    \item Reinício do buffer e continuação da interação com política atualizada.
\end{itemize}
\end{frame}

\section{Resultados}
\begin{frame}{Resultados principais}
\begin{itemize}
    \item Observação de fases distintas no processo de aprendizado.
    \item Tendências no plano de informação indicam potencial convergência.
    \item Diferenças entre topologias afetam dinâmica e estabilidade.
    \item Flutuações na IM estimada são inerentes ao método de estimação.
\end{itemize}
\end{frame}

\section{Conclusão}
\begin{frame}{Conclusoes e contribuicoes}
\begin{itemize}
    \item O plano de informação é uma métrica alternativa promissora para análise em RL.
    \item Há evidências de estágios distintos ao longo do treinamento.
    \item Generalização dos padrões ainda requer investigação aprofundada.
    \item As trajetórias no plano de informação trazem sinais úteis para diagnóstico do aprendizado.
\end{itemize}
\end{frame}

\begin{frame}{Proximos passos}
\begin{itemize}
    \item Expandir análise para ambientes mais complexos da Gymnasium.
    \item Comparar com DQN, SAC e TD3.
    \item Investigar impacto de hiperparâmetros e topologias na trajetória da IM.
    \item Explorar uso da IM como critério de convergência do agente.
\end{itemize}
\end{frame}
